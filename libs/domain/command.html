191    def __init__(self) -> None:
192        self.day_format = re.compile(r"^([0-9]{8}|[0-9/.-]{8,10})$")
193        """日付文字列判定用正規表現
194        - *yyyymmdd*
195        - *yyyy/mm/dd*, *yyyy/m/d*
196        - *yyyy-mm-dd*, *yyyy-m-d*
197        - *yyyy.mm.dd*, *yyyy.m.d*
198        """

200    @classmethod
201    def is_valid_command(cls, word: str) -> bool:
202        """
203        引数がコマンド名と一致するか判定する
204
205        Args:
206            word (str): チェック文字列
207
208        Returns:
209            bool: 真偽
210
211        """
212        for cmd in COMMANDS.values():
213            for pattern in cmd["match"]:
214                m = re.match(pattern, word)
215                if m:
216                    return True
217                m = re.match(pattern, textutil.str_conv(word.lower(), textutil.ConversionType.HtoK))
218                if m:
219                    return True
220
221        return False
222
223    def analysis_argument(self, argument: list[str]) -> ParsedCommand:
224        """
225        コマンドライン引数を解析する
226
227        Args:
228            argument (list[str]): 引数
229
230        Returns:
231            ParsedCommand: 結果
232
233        """
234        ret: dict[str, Any] = {}
235        unknown: list[str] = []
236        args: list[str] = []
237        search_range: list[Any] = []
238
239        for keyword in argument:
240            check_word = textutil.str_conv(keyword.lower(), textutil.ConversionType.HtoK)
241            check_word = check_word.replace("無シ", "ナシ").replace("有リ", "アリ")
242
243            if re.match(r"^([0-9]{8}|[0-9/.-]{8,10})$", check_word):
244                try_day = pd.to_datetime(check_word, errors="coerce").to_pydatetime()
245                if not pd.isna(try_day):
246                    search_range.append(ExtDt(try_day))
247                    search_range.append(ExtDt(try_day) + {"hour": 23, "minute": 59, "second": 59, "microsecond": 999999})
248                continue
249
250            if check_word in ExtDt.valid_keywords():
251                search_range.append(check_word)
252                continue
253
254            for cmd in COMMANDS.values():
255                for pattern in cmd["match"]:
256                    m = re.match(pattern, keyword)
257                    if m:
258                        ret.update(self._parse_match(cmd, m))
259                        break
260                    m = re.match(pattern, check_word)
261                    if m:
262                        ret.update(self._parse_match(cmd, m))
263                        break
264                else:
265                    continue
266                break
267            else:
268                unknown.append(formatter.name_replace(keyword, add_mark=False, not_replace=True))
269
270        return ParsedCommand(flags=ret, arguments=args, unknown=unknown, search_range=search_range)
271
272    def _parse_match(self, cmd: CommandSpec, obj: re.Match) -> dict[str, Any]:
273        """
274        コマンド名に一致したときの処理
275
276        Args:
277            cmd (CommandSpec): コマンドマップ
278            obj (re.Match): Matchオブジェクト
279
280        Returns:
281            dict[str, Any]: 更新用辞書
282
283        """
284        ret: dict[str, Any] = {}
285
286        def with_arguments(tmp: dict[str, Any]) -> None:
287            key = str(next(iter(tmp.keys())))
288            val = str(tmp[key][1])
289            if "" != val:
290                match cmd.get("type"):
291                    case "str":
292                        ret.update({key: val})
293                    case "sql":
294                        ret.update({key: f"%{val}%"})
295                    case "filename":
296                        if re.search(r"^[\w\-\.]+$", val):
297                            ret.update({key: val})
298                    case "int":
299                        ret.update({key: int(val)})
300                    case _:
301                        ret.update({key: int(val) if val.isdigit() else val})
302
303        match len(obj.groups()):
304            case 0:  # 完全一致: ^command$
305                ret.update(cmd["action"](obj.group()))
306            case 1:  # 選択: ^(command1|command2|...)$
307                ret.update(cmd["action"](obj.groups()[0]))
308            case 2:  # 引数あり: ^(command)(\d*)$
309                tmp = cmd["action"](obj.groups())
310                if isinstance(tmp, dict):
311                    for k, v in tmp.items():
312                        if isinstance(v, tuple):  # 引数取り出し&セット
313                            with_arguments(tmp)
314                        if isinstance(v, bool):  # フラグ上書き
315                            ret.update({k: v})
316
317        return ret

436    def placeholder(self, game_count: Optional[int] = None) -> dict[str, Any]:
437        """
438        プレースホルダ用辞書出力
439
440        Args:
441            game_count (Optional[int]): 規定打数調整用ゲーム数. Defaults to None.
442
443        Returns:
444            dict[str, Any]: プレースホルダ
445
446        """
447        ret_dict: dict[str, Any] = asdict(self)
448
449        # 規定打数更新
450        if not ret_dict.get("stipulated") or game_count is not None:
451            if game_count is None:
452                ret_dict.update({"stipulated": 1})
453            else:
454                ret_dict.update({"stipulated": int(ceil(game_count * self.stipulated_rate) + 1)})
455
456        if self.player_list:
457            ret_dict.update({f"player_{idx}": x for idx, x in enumerate(self.player_list)})
458        ret_dict.pop("player_list")
459
460        if self.target_player:
461            ret_dict.update({f"target_{idx}": x for idx, x in enumerate(self.target_player)})
462        ret_dict.pop("target_player")
463
464        if self.competition_list:
465            ret_dict.update({f"competition_{idx}": x for idx, x in enumerate(self.competition_list)})
466        ret_dict.pop("competition_list")
467
468        if self.rule_list:
469            ret_dict.update({f"rule_{idx}": x for idx, x in enumerate(self.rule_list)})
470        else:
471            ret_dict.update({"rule_0": self.rule_version})
472        ret_dict.pop("rule_list")
473
474        return ret_dict

273    def _parse_match(self, cmd: CommandSpec, obj: re.Match) -> dict[str, Any]:
274        """
275        コマンド名に一致したときの処理
276
277        Args:
278            cmd (CommandSpec): コマンドマップ
279            obj (re.Match): Matchオブジェクト
280
281        Returns:
282            dict[str, Any]: 更新用辞書
283
284        """
285        ret: dict[str, Any] = {}
286
287        def with_arguments(tmp: dict[str, Any]) -> None:
288            key = str(next(iter(tmp.keys())))
289            val = str(tmp[key][1])
290            if "" != val:
291                match cmd.get("type"):
292                    case "str":
293                        ret.update({key: val})
294                    case "sql":
295                        ret.update({key: f"%{val}%"})
296                    case "filename":
297                        if re.search(r"^[\w\-\.]+$", val):
298                            ret.update({key: val})
299                    case "int":
300                        ret.update({key: int(val)})
301                    case _:
302                        ret.update({key: int(val) if val.isdigit() else val})
303
304        match len(obj.groups()):
305            case 0:  # 完全一致: ^command$
306                ret.update(cmd["action"](obj.group()))
307            case 1:  # 選択: ^(command1|command2|...)$
308                ret.update(cmd["action"](obj.groups()[0]))
309            case 2:  # 引数あり: ^(command)(\d*)$
310                tmp = cmd["action"](obj.groups())
311                if isinstance(tmp, dict):
312                    for k, v in tmp.items():
313                        if isinstance(v, tuple):  # 引数取り出し&セット
314                            with_arguments(tmp)
315                        if isinstance(v, bool):  # フラグ上書き
316                            ret.update({k: v})
317
318        return ret
