 65class BaseSection(CommonMethodMixin):
 66    """基底クラス"""
 67
 68    section: str
 69    """セクション名"""
 70    main_parser: "ConfigParser"
 71    """設定パーサー"""
 72    section_proxy: "SectionProxy"
 73    """読み込み先(パーサー + セクション名)"""
 74
 75    def __init__(self, outer: SubClassType) -> None:
 76        self.main_parser = outer.main_parser
 77        assert self.main_parser
 78        if not hasattr(self, "section") or self.section not in self.main_parser:
 79            return
 80        self.section_proxy = self.main_parser[self.section]
 81
 82        self.initialization()
 83
 84    def __repr__(self) -> str:
 85        return str({k: v for k, v in vars(self).items() if not str(k).startswith("_")})
 86
 87    def initialization(self) -> None:
 88        """設定ファイルから値の取り込み"""
 89        for k in self.section_proxy.keys():
 90            if k not in self.to_dict():
 91                continue  # インスタンス変数と一致しない項目はスキップ
 92            match type(self.__dict__.get(k)):
 93                case v_type if k in self.__dict__ and v_type is str:
 94                    setattr(self, k, self.get(k))
 95                case v_type if k in self.__dict__ and v_type is int:
 96                    setattr(self, k, self.getint(k))
 97                case v_type if k in self.__dict__ and v_type is float:
 98                    setattr(self, k, self.getfloat(k))
 99                case v_type if v_type is bool:
100                    setattr(self, k, self.section_proxy.getboolean(k))
101                case v_type if k in self.__dict__ and v_type is list:
102                    v_list = self.getlist(k)
103                    current_list = getattr(self, k)
104                    if isinstance(current_list, list) and current_list:  # 設定済みリストは追加
105                        current_list.extend(v_list)
106                    else:
107                        setattr(self, k, v_list)
108                case v_type if k in self.__dict__ and get_origin(v_type) is Union and type(None) in get_args(v_type):  # 文字列 or None(未定義)
109                    setattr(self, k, self.get(k))
110                case v_type if k in self.__dict__ and v_type is PosixPath:
111                    setattr(self, k, Path(self.get(k)))
112                case v_type if k in self.__dict__ and v_type is NoneType:
113                    if k in ["backup_dir"]:  # ディレクトリを指定する設定はPathで格納
114                        setattr(self, k, Path(self.get(k)))
115                    else:
116                        setattr(self, k, self.get(k))
117                case _:
118                    setattr(self, k, self.__dict__.get(k))
119
120    def to_dict(self, drop_items: Optional[list[str]] = None) -> dict[str, str]:
121        """
122        必要なパラメータを辞書型で返す
123
124        Args:
125            drop_items (Optional[list[str]], optional): 返却に含めないキーリスト. Defaults to None.
126
127        Returns:
128             dict[str, str]: 返却値
129
130        """
131        ret_dict: dict[str, str] = {}
132        for key in vars(self):
133            if key.startswith("_"):
134                continue
135            ret_dict[key] = getattr(self, key)
136
137        if drop_items:
138            for item in drop_items:
139                if item in ret_dict:
140                    ret_dict.pop(item)
141
142        return ret_dict

159        """順位点"""

174        """
175        設定値取り込み
176
177        Args:
178            outer (AppConfig): 設定クラスオブジェクト
179
180        """
181        self.main_parser = outer.main_parser
182        super().__init__(self)
183
184        # デフォルト値
185        match self.mode:
186            case 3:
187                if self.origin_point < 0:
188                    self.origin_point = 350
189                if self.return_point < 0:
190                    self.return_point = 400
191                if not self.rank_point:
192                    self.rank_point = [30, 0, -30]
193            case 4:
194                if self.origin_point < 0:
195                    self.origin_point = 250
196                if self.return_point < 0:
197                    self.return_point = 300
198                if not self.rank_point:
199                    self.rank_point = [30, 10, -10, -30]
200
201        self.rank_point = self.rank_point[: self.mode]
202
203        logging.debug("%s: %s", self.section, self)

266    def config_load(self, outer: "AppConfig") -> None:
267        """
268        設定値取り込み
269
270        Args:
271            outer (AppConfig): 設定クラスオブジェクト
272
273        """
274        self.main_parser = outer.main_parser
275        self._reset()
276        super().__init__(self)
277
278        # 成績登録キーワード
279        if not (isinstance(self.keyword, Path) and self.keyword.exists()):
280            self.keyword = str(self.keyword)

283        if not self.default_rule:

286        # フォントファイルチェック
287        for chk_dir in (outer.config_dir, outer.script_dir):
288            chk_file = chk_dir / str(self.font_file)
289            if chk_file.exists():
290                self.font_file = chk_file
291                break
292        else:
293            if not self.font_file.exists():
294                logging.critical("The specified font file cannot be found.")
295                sys.exit(255)
296
297        # 作業ディレクトリパス
298        if not self.work_dir.is_absolute():
299            self.work_dir = outer.script_dir / self.work_dir
300
301        # データベース関連
302        if isinstance(self.database_file, Path) and not self.database_file.exists():
303            self.database_file = outer.config_dir / str(self.database_file)

 92                continue  # インスタンス変数と一致しない項目はスキップ
 93            match type(self.__dict__.get(k)):
 94                case v_type if k in self.__dict__ and v_type is str:
 95                    setattr(self, k, self.get(k))
 96                case v_type if k in self.__dict__ and v_type is int:
 97                    setattr(self, k, self.getint(k))
 98                case v_type if k in self.__dict__ and v_type is float:
 99                    setattr(self, k, self.getfloat(k))
100                case v_type if v_type is bool:
101                    setattr(self, k, self.section_proxy.getboolean(k))
102                case v_type if k in self.__dict__ and v_type is list:
103                    v_list = self.getlist(k)
104                    current_list = getattr(self, k)
105                    if isinstance(current_list, list) and current_list:  # 設定済みリストは追加
106                        current_list.extend(v_list)
107                    else:
108                        setattr(self, k, v_list)
109                case v_type if k in self.__dict__ and get_origin(v_type) is Union and type(None) in get_args(v_type):  # 文字列 or None(未定義)
110                    setattr(self, k, self.get(k))
111                case v_type if k in self.__dict__ and v_type is PosixPath:
112                    setattr(self, k, Path(self.get(k)))
113                case v_type if k in self.__dict__ and v_type is NoneType:
114                    if k in ["backup_dir"]:  # ディレクトリを指定する設定はPathで格納
115                        setattr(self, k, Path(self.get(k)))
116                    else:
117                        setattr(self, k, self.get(k))
118                case _:
119                    setattr(self, k, self.__dict__.get(k))
