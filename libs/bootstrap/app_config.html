387    def word_list(self) -> list[str]:
388        """設定されている値、キーワードをリスト化する

411    def overwrite(self, additional_config: Path, section_name: str):
412        """指定セクションを上書き
413
414        Args:
415            additional_config (Path): 追加設定ファイルパス
416            section_name (str): セクション名
417        """
418
419        if not additional_config.exists():
420            return
421
422        try:
423            self._parser = ConfigParser()
424            self._parser.read([self.config_file, additional_config], encoding="utf-8")
425        except Exception as err:
426            logging.error(err)
427            return
428
429        protected_values: Union[str, list]
430        match section_name:
431            case "setting":
432                protected_values = self.setting.help  # 上書き保護
433                self.setting.config_load(self)
434                self.setting.help = protected_values
435            case "results":
436                protected_values = self.results.commandword  # 上書き保護
437                self.results.config_load(self)
438                self.results.commandword = protected_values
439            case "graph":
440                protected_values = self.graph.commandword  # 上書き保護
441                self.graph.config_load(self)
442                self.graph.commandword = protected_values
443            case "ranking":
444                protected_values = self.ranking.commandword  # 上書き保護
445                self.ranking.config_load(self)
446                self.ranking.commandword = protected_values
447            case "report":
448                protected_values = self.report.commandword  # 上書き保護
449                self.report.config_load(self)
450                self.report.commandword = protected_values
451            case _:
452                return

454    def read_channel_config(self, section_name: str) -> Optional[Path]:
455        """チャンネル個別設定読み込み
456
457        Args:
458            section_name (str): セクション名
459
460        Returns:
461            Optional[Path]: 個別設定読み込み結果
462                - *Path*: 読み込んだ設定ファイルパス
463                - *None*: 読み込める設定ファイルがない
464        """
465
466        config_path: Optional[Path] = None
467        self.initialization()
468
469        if self.main_parser.has_section(section_name):
470            if channel_config := self.main_parser[section_name].get("channel_config"):
471                config_path = Path(channel_config)
472                if config_path.exists():
473                    logging.debug("Override: %s", config_path.absolute())
474                    self.overwrite(config_path, "setting")
475                    self.overwrite(config_path, "results")
476                    self.overwrite(config_path, "graph")
477                    self.overwrite(config_path, "ranking")
478                    self.overwrite(config_path, "report")
479                else:
480                    config_path = None
481
482        read_memberslist()
483
484        return config_path

486    def resolve_channel_id(self, section_name: Optional[str] = None) -> str:
487        """メイン設定から優先度の高いチャンネルIDを取得する
488
489        Args:
490            section_name (Optional[str]): チャンネル個別設定セクション名
491
492        Returns:
493            str: チャンネルID
494        """
495
496        for section in (section_name, self.selected_service, "setting"):
497            if section and self.main_parser.has_section(section):
498                if channel_id := self.main_parser[section].get("channel_id"):
499                    return channel_id
500
501        if section_name:
502            return section_name
503        return ""
